# 本机接口（HTTP）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**目标**：第三方网页和客户端软件通过本机或局域网的 HTTP 接口提交「模板 + 字段」，程序排版、按纸张决定打印机并打印，或者返回 PDF；所有打印记录（包括扫码枪和手机）完整保存字段，供以后分析和导出。

**架构**：
- 接口服务在主进程里，用 `node:http`，不加依赖。
- 纯逻辑放在不依赖 Electron 的文件里，用 `bun test` 测试：
  - 任务服务 `src/core/api/`；
  - HTTP 处理、授权、校验 `src/main/api/`。
- 打印走第 1 个子项目的 `PrintService.printLabel`，对外新增 `printFields`。
- 接口遵循 Google AIP，浏览器访问遵循 Chrome 的本地网络访问规则。

**技术栈**：TypeScript、Electron（`printToPDF`）、`node:http`、`node:crypto`、SQLite（`node:sqlite`）、React 19、bun test、Playwright、Biome、NSIS。

**设计文档**：`docs/superpowers/specs/2026-09-30-local-api-design.md`

**约定**：
- 测试先行，每个提交都通过 `bun run check`，提交前 `bun run lint:fix`。
- 改了界面或主进程再跑 `bun run test:e2e`；改了安装包跑 `bun run dist:win` 并实际装、更新、卸载一次。
- 注释中文，标识符和提交信息英文；常量带单位和取值依据。
- 新代码遵循 Google TypeScript 风格指南，和仓库约定冲突时以仓库为准：
  - 只用具名导出；
  - 不用 `any`；
  - 类型 UpperCamelCase，常量 CONSTANT_CASE；
  - 能 `readonly` 的都 `readonly`。
- 只在一个平台验证过的改动，在提交说明里写明。
- 中转服务相关代码有改动时直接部署（已授权）；本计划不改中转服务。

```ts
  async printFields(input: FieldsPrint): Promise<PrintResult> {
```

## 阶段 2：任务服务（纯逻辑）

**目标**：`PrintJobService` 能建任务、按 `requestId` 防重复、逐份打印、更新状态、限制排队、清理过期任务、程序重启后收尾。全部是纯逻辑，不依赖 Node 和 Electron。

**状态**：Complete（8e714da、d150661）。执行时和计划不同的地方：
- 任务不各自并发执行，而是用 `SerialQueue` 排成一队按提交顺序打（设计文档第 6 节已同步）：并发会把所有任务同时标成打印中，不同任务的份数也会交错出纸。
- 同一批里重复的 `requestId` 只建一个任务；打印服务意外抛错时任务落到 `FAILED`，不卡在打印中。
- 存储的清理方法叫 `deleteFinishedBefore`，只删已结束的任务；`list` 按调用方过滤（调用方只能看到自己的任务）。
- 由字段拼成的完整内容截到 1000 个字，和扫码内容的上限一致。

```ts
const COLLECTION = 'templates/';

/** 程序里的模板编号（builtin:standard）→ 接口里的名字（templates/builtin-standard）。 */
export function templateName(templateId: string): string {

/** 接口里的名字或编号（templates/builtin-standard、builtin-standard）→ 程序里的模板编号；不是模板名时为 null。 */
export function templateIdFromName(name: string): string | null {
  const id = name.startsWith(COLLECTION) ? name.slice(COLLECTION.length) : name;
  for (const prefix of PREFIXES) {
    if (id.startsWith(`${prefix}-`) && id.length > prefix.length + 1) {
      return `${prefix}:${id.slice(prefix.length + 1)}`;
    }
  }
  return null;
}
```

```ts
import type { LabelTemplate } from '../templates/template-model';
import type { Clock, PrintResult } from '../types';
import type { FieldsPrint } from '../print-service';
import { contentOf, type PrintJob, type PrintJobFailure, type PrintJobInput } from './api-model';
import type { ApiJobStore } from './api-job-store';

/** AIP-155：同一调用方的同一 requestId 在这段时间内只处理一次。 */
export const REQUEST_ID_WINDOW_MS = 24 * 60 * 60_000;
/** 任务状态保留 7 天：调用方查询用；标签内容以打印记录为准。 */
export const JOB_RETENTION_MS = 7 * 24 * 60 * 60_000;

export type PrintJobErrorCode = 'TEMPLATE_NOT_FOUND' | 'PRINTER_NOT_FOUND' | 'QUEUE_FULL';

/** 提交时就能判断的错误：一张都不排队。index 是批量请求里的第几个。 */
export class PrintJobError extends Error {
  constructor(
    readonly code: PrintJobErrorCode,
    message: string,
    readonly index: number | null = null,
  ) {
    super(message);
    this.name = 'PrintJobError';
  }
}

export interface PrintJobServiceDeps {
  store: ApiJobStore;
  clock: Clock;
  createId: () => string;
  /** 按编号精确查找（找不到为 null，不退回通用模板）；不叠加工作台的备注选择。 */
  findTemplate: (templateId: string) => LabelTemplate | null;
  installedPrinters: () => Promise<string[]>;
  printFields: (input: FieldsPrint) => Promise<PrintResult>;
  /** 排队中的标签总数上限：防止调用方出错时刷爆打印机。 */
  queueLimit: number;
}

const FAILURE_MESSAGES: Record<PrintJobFailure['reason'], string> = {
  NO_PRINTER: '这种纸还没有打印机：请在电脑上为这种纸指定打印机',
  PRINTER_NOT_FOUND: '打印机不在这台电脑上：请检查打印机是否连接',
  PRINTER_NOT_READY: '打印机不能打印（缺纸、离线或卡纸）',
  PRINT_TIMEOUT: '打印超时：可能已出纸，请到打印机旁确认',
  PRINT_ERROR: '打印机驱动报告错误',
  INTERRUPTED: '程序在打完之前退出了：已发送的份数见 sentCopies，其余没有打印',
};

export class PrintJobService {
  private running: Promise<void>[] = [];

  constructor(private readonly deps: PrintJobServiceDeps) {}

  async create(caller: string, input: PrintJobInput): Promise<PrintJob> {
    const [job] = await this.createBatch(caller, [input]);
    if (!job) {
      throw new Error('createBatch returned no job for one request');
    }
    return job;
  }

  /** 整批先核对（模板、打印机、排队上限），有一个不行就一张都不收；收下后每张各自打印。 */
  async createBatch(caller: string, inputs: readonly PrintJobInput[]): Promise<PrintJob[]> {
    const now = this.deps.clock.now();
    const installed = await this.deps.installedPrinters();
    const existing = inputs.map((input) =>
      input.requestId === null
        ? null
        : this.deps.store.findByRequestId(caller, input.requestId, now - REQUEST_ID_WINDOW_MS),
    );
    let newLabels = 0;
    inputs.forEach((input, index) => {
      if (existing[index]) {
        return;
      }
      if (this.deps.findTemplate(input.templateId) === null) {
        throw new PrintJobError('TEMPLATE_NOT_FOUND', `找不到模板 ${input.templateId}`, index);
      }
      if (input.printer !== null && !installed.includes(input.printer)) {
        throw new PrintJobError('PRINTER_NOT_FOUND', `这台电脑上没有打印机「${input.printer}」`, index);
      }
      newLabels += input.copies;
    });
    if (this.deps.store.pendingLabels() + newLabels > this.deps.queueLimit) {
      throw new PrintJobError('QUEUE_FULL', `排队中的标签太多（上限 ${this.deps.queueLimit} 张），请稍后再提交`);
    }
    return inputs.map((input, index) => existing[index] ?? this.enqueue(caller, input, now));
  }

  get(id: string): PrintJob | null {
    return this.deps.store.get(id);
  }

  /** 等所有正在处理的任务结束（测试和退出时用）。 */
  async idle(): Promise<void> {
    while (this.running.length > 0) {
      await Promise.all(this.running);
    }
  }

  purge(): void {
    this.deps.store.deleteBefore(this.deps.clock.now() - JOB_RETENTION_MS);
  }

  /** 启动时调用：上次没打完的任务标成失败，不自动续打（免得突然出纸）。 */
  recoverInterrupted(): void {
    for (const job of this.deps.store.listUnfinished()) {
      this.save(job, { state: 'FAILED', failure: { reason: 'INTERRUPTED', message: FAILURE_MESSAGES.INTERRUPTED } });
    }
  }

  private enqueue(caller: string, input: PrintJobInput, now: number): PrintJob {
    const job: PrintJob = {
      ...input,
      id: this.deps.createId(),
      caller,
      state: 'QUEUED',
      sentCopies: 0,
      failure: null,
      createdAt: now,
      updatedAt: now,
    };
    this.deps.store.insert(job);
    const run = this.run(job).finally(() => {
      this.running = this.running.filter((item) => item !== run);
    });
    this.running.push(run);
    return job;
  }

  /** 逐份打印：某一份失败就停，后面的份数不再打。 */
  private async run(queued: PrintJob): Promise<void> {
    let job = this.save(queued, { state: 'PRINTING' });
    const template = this.deps.findTemplate(job.templateId);
    if (template === null) {
      this.save(job, { state: 'FAILED', failure: { reason: 'PRINT_ERROR', message: '模板在打印前被删除了' } });
      return;
    }
    for (let copy = job.sentCopies; copy < job.copies; copy += 1) {
      const result = await this.deps.printFields({
        template,
        fields: job.fields,
        content: contentOf(job),
        source: 'api',
        caller: job.caller,
        printerName: job.printer,
      });
      const failure = failureOf(result);
      if (failure) {
        this.save(job, { state: 'FAILED', failure });
        return;
      }
      job = this.save(job, { sentCopies: job.sentCopies + 1 });
    }
    this.save(job, { state: 'SENT' });
  }

  private save(job: PrintJob, patch: Partial<PrintJob>): PrintJob {
    const next = { ...job, ...patch, updatedAt: this.deps.clock.now() };
    this.deps.store.update(next);
    return next;
  }
}

function failureOf(result: PrintResult): PrintJobFailure | null {
  switch (result.status) {
    case 'printed':
      return null;
    case 'no-printer':
      return { reason: 'NO_PRINTER', message: FAILURE_MESSAGES.NO_PRINTER };
    case 'failed': {
      const reason = result.reason === 'LOOKUP_FAILED' ? 'PRINT_ERROR' : result.reason;
      return { reason, message: result.detail ?? FAILURE_MESSAGES[reason] };
    }
    // 本机接口不走扫码防重复、不走识别规则：这两种不会出现，按驱动错误处理，不让任务卡住。
    case 'duplicate':
    case 'invalid':
      return { reason: 'PRINT_ERROR', message: FAILURE_MESSAGES.PRINT_ERROR };
```

```ts
/** AIP-193 的规范错误码 → HTTP 状态码（google.rpc.Code 的 HTTP 映射）。 */
const HTTP_STATUS = {
  INVALID_ARGUMENT: 400,
  FAILED_PRECONDITION: 400,
  UNAUTHENTICATED: 401,
  PERMISSION_DENIED: 403,
  NOT_FOUND: 404,
  ALREADY_EXISTS: 409,
  RESOURCE_EXHAUSTED: 429,
  INTERNAL: 500,
  UNAVAILABLE: 503,
} as const;
export type CanonicalCode = keyof typeof HTTP_STATUS;

/** 413 不是 google.rpc.Code：请求体超限按 INVALID_ARGUMENT 报，HTTP 状态用 413，方便调用方识别。 */
const PAYLOAD_TOO_LARGE = 413;
const ERROR_DOMAIN = 'labelflash';

export interface FieldViolation {
  field: string;
  description: string;
}

export class ApiError extends Error {
  readonly httpStatus: number;

  constructor(
    readonly status: CanonicalCode,
    readonly reason: string,
    message: string,
    readonly fieldViolations: readonly FieldViolation[] = [],
    httpStatus?: number,
  ) {
    super(message);
    this.name = 'ApiError';
    this.httpStatus = httpStatus ?? HTTP_STATUS[status];
  }

  static invalidArgument(message: string, violations: readonly FieldViolation[] = []): ApiError {
    return new ApiError('INVALID_ARGUMENT', 'INVALID_ARGUMENT', message, violations);
  }

  static payloadTooLarge(limitBytes: number): ApiError {
    return new ApiError('INVALID_ARGUMENT', 'PAYLOAD_TOO_LARGE', `请求体超过 ${limitBytes} 字节：请分批提交`, [], PAYLOAD_TOO_LARGE);
  }
}

export function errorBody(error: ApiError): unknown {
  const details: unknown[] = [
    { '@type': 'type.googleapis.com/google.rpc.ErrorInfo', reason: error.reason, domain: ERROR_DOMAIN },
  ];
  if (error.fieldViolations.length > 0) {
    details.push({ '@type': 'type.googleapis.com/google.rpc.BadRequest', fieldViolations: error.fieldViolations });
  }
  return { error: { code: error.httpStatus, status: error.status, message: error.message, details } };
}
```

```ts
import type { PrintJobInput } from '../../core/api/api-model';
import { templateIdFromName } from '../../core/api/template-names';
import { isValidFieldName } from '../../core/scan/rule-model';
import { MAX_RAW_LENGTH } from '../../core/scan/normalize-raw';
import type { ScanField } from '../../core/scan/scan-result';
import { ApiError, type FieldViolation } from './api-error';

/** 一次批量最多 1000 张：服装标签一次两三百张很常见，再多由调用方分批。 */
export const BATCH_LIMIT = 1000;
/** 一张标签最多 50 个字段：面单的字段最多二三十个，留出余量。 */
export const FIELDS_LIMIT = 50;
/** 份数上限：同一张标签打这么多份已经远超日常需要。 */
export const COPIES_LIMIT = 100;
const PRINTER_NAME_LENGTH = 256;
/** AIP-155：requestId 是 UUID。 */
const UUID_PATTERN = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

type Loose = Record<string, unknown>;

export function parseCreateRequest(body: unknown): PrintJobInput {
  const violations: FieldViolation[] = [];
  const input = readJob(body, '', violations);
  if (violations.length > 0 || input === null) {
    throw ApiError.invalidArgument('请求参数不对，见 fieldViolations', violations);
  }
  return input;
}

export function parseBatchRequest(body: unknown): PrintJobInput[] {
  const requests = isLoose(body) ? body['requests'] : undefined;
  if (!Array.isArray(requests) || requests.length === 0 || requests.length > BATCH_LIMIT) {
    throw ApiError.invalidArgument(`requests 必须是 1–${BATCH_LIMIT} 个请求`, [
      { field: 'requests', description: `1–${BATCH_LIMIT} 个` },
    ]);
  }
  const violations: FieldViolation[] = [];
  const inputs = requests.map((request, index) => readJob(request, `requests[${index}].`, violations));
  if (violations.length > 0) {
    throw ApiError.invalidArgument('有请求的参数不对，整批都没有收下，见 fieldViolations', violations);
  }
  return inputs.filter((input): input is PrintJobInput => input !== null);
}

export function parseRenderRequest(body: unknown): { fields: ScanField[]; content: string | null } {
  const violations: FieldViolation[] = [];
  const input = isLoose(body) ? body : {};
  const fields = readFields(input['fields'], 'fields', violations);
  const content = readContent(input['content'], 'content', violations);
  if (violations.length > 0) {
    throw ApiError.invalidArgument('请求参数不对，见 fieldViolations', violations);
  }
  return { fields, content };
}

function readJob(value: unknown, prefix: string, violations: FieldViolation[]): PrintJobInput | null {
  const input = isLoose(value) ? value : {};
  const before = violations.length;
  const template = typeof input['template'] === 'string' ? templateIdFromName(input['template']) : null;
  if (template === null) {
    violations.push({ field: `${prefix}template`, description: '模板名，例如 templates/builtin-standard' });
  }
  const fields = readFields(input['fields'], `${prefix}fields`, violations);
  const content = readContent(input['content'], `${prefix}content`, violations);
  const copies = input['copies'] ?? 1;
  if (typeof copies !== 'number' || !Number.isInteger(copies) || copies < 1 || copies > COPIES_LIMIT) {
    violations.push({ field: `${prefix}copies`, description: `1–${COPIES_LIMIT} 的整数` });
  }
  const printer = input['printer'] ?? null;
  if (printer !== null && (typeof printer !== 'string' || printer === '' || printer.length > PRINTER_NAME_LENGTH)) {
    violations.push({ field: `${prefix}printer`, description: '系统里的打印机名' });
  }
  const requestId = input['requestId'] ?? null;
  if (requestId !== null && (typeof requestId !== 'string' || !UUID_PATTERN.test(requestId))) {
    violations.push({ field: `${prefix}requestId`, description: 'UUID' });
  }
  if (violations.length > before || template === null) {
    return null;
  }
  return {
    templateId: template,
    fields,
    content,
    copies: copies as number,
    printer: printer as string | null,
    requestId: requestId as string | null,
  };
}

function readFields(value: unknown, path: string, violations: FieldViolation[]): ScanField[] {
  if (!Array.isArray(value) || value.length === 0 || value.length > FIELDS_LIMIT) {
    violations.push({ field: path, description: `1–${FIELDS_LIMIT} 个 { name, value }` });
    return [];
  }
  const seen = new Set<string>();
  const fields: ScanField[] = [];
  value.forEach((item, index) => {
    const name = isLoose(item) ? item['name'] : undefined;
    const fieldValue = isLoose(item) ? item['value'] : undefined;
    if (typeof name !== 'string' || !isValidFieldName(name) || seen.has(name)) {
      violations.push({ field: `${path}[${index}].name`, description: '1–20 个字，不能有花括号和控制字符，不能重名' });
      return;
    }
    if (typeof fieldValue !== 'string' || fieldValue.length > MAX_RAW_LENGTH) {
      violations.push({ field: `${path}[${index}].value`, description: `字符串，最长 ${MAX_RAW_LENGTH} 个字` });
      return;
    }
    seen.add(name);
    fields.push({ name, value: fieldValue });
  });
  return fields;
}

function readContent(value: unknown, path: string, violations: FieldViolation[]): string | null {
  if (value === undefined || value === null) {
    return null;
  }
  if (typeof value !== 'string' || value === '' || value.length > MAX_RAW_LENGTH) {
    violations.push({ field: path, description: `字符串，1–${MAX_RAW_LENGTH} 个字` });
    return null;
  }
  return value;
}

function isLoose(value: unknown): value is Loose {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}
```

```ts
import { isIP } from 'node:net';

/** 对方是不是本机：127.0.0.0/8、::1、IPv4 映射的 127.x。 */
export function isLoopbackAddress(address: string | undefined): boolean {
  if (address === undefined) {
    return false;
  }
  const plain = address.startsWith('::ffff:') ? address.slice('::ffff:'.length) : address;
  return plain === '::1' || (isIP(plain) === 4 && plain.startsWith('127.'));
}

const LOOPBACK_HOSTNAMES = new Set(['127.0.0.1', 'localhost', '[::1]']);

/** 网页请求的 Host 必须是本机名加本服务的端口（防 DNS 重绑定）。 */
export function isLoopbackHost(host: string | undefined, port: number): boolean {
  if (host === undefined) {
    return false;
  }
  const separator = host.lastIndexOf(':');
  if (separator <= 0) {
    return false;
  }
  return LOOPBACK_HOSTNAMES.has(host.slice(0, separator).toLowerCase()) && host.slice(separator + 1) === String(port);
}
```

```ts
  constructor(private readonly deps: AuthenticatorDeps) {}

  authenticate(request: AuthRequest): Caller {
    const authorization = request.headers['authorization'];
    if (authorization?.startsWith(BEARER)) {
      const key = this.deps.findKeyByHash(hashApiKey(authorization.slice(BEARER.length).trim()));
      if (key === null) {
        throw new ApiError('UNAUTHENTICATED', 'KEY_INVALID', '程序密钥不对或已撤销：请在电脑上的「本机接口」页查看');
      }
      this.deps.touchKey(key.id);
      return { id: `key:${key.id}`, label: key.name };
    }
    const origin = request.headers['origin'];
    // 网站授权只对本机来的浏览器请求：Origin 由浏览器填写；局域网来的请求一律要密钥。
    if (origin !== undefined && isLoopbackAddress(request.remoteAddress)) {
      if (!isLoopbackHost(request.headers['host'], request.port)) {
        throw new ApiError('PERMISSION_DENIED', 'HOST_NOT_ALLOWED', '请用 http://127.0.0.1 访问本机接口');
      }
      if (this.deps.isOriginAuthorized(origin)) {
        return { id: `origin:${origin}`, label: origin };
      }
      this.deps.requestOrigin(origin);
      throw new ApiError('PERMISSION_DENIED', 'ORIGIN_NOT_AUTHORIZED', '请在电脑上点「允许」，然后重试');
    }
    if (!this.deps.hasAnyKey()) {
      throw new ApiError('UNAUTHENTICATED', 'NO_KEYS_YET', '还没有程序密钥：请先在电脑上的「本机接口」页生成');
    }
    throw new ApiError('UNAUTHENTICATED', 'KEY_REQUIRED', '请在 Authorization 请求头里带上程序密钥（Bearer）');
  }
}
```

```ts
import type { Clock } from '../../core/types';

/** 用户拒绝某网站后这么久不再弹框：网页多半会自动重试，不能一直打扰。 */
export const DENIAL_COOLDOWN_MS = 10 * 60_000;

export interface OriginPromptsDeps {
  clock: Clock;
  /** 弹系统对话框，返回是否允许。 */
  ask: (origin: string) => Promise<boolean>;
  /** 允许：记进设置。 */
  grant: (origin: string) => void;
}

export class OriginPrompts {
  private readonly pending = new Map<string, Promise<void>>();
  private readonly deniedAt = new Map<string, number>();

  constructor(private readonly deps: OriginPromptsDeps) {}

  request(origin: string): void {
    if (this.pending.has(origin)) {
      return;
    }
    const denied = this.deniedAt.get(origin);
    if (denied !== undefined && this.deps.clock.now() - denied < DENIAL_COOLDOWN_MS) {
      return;
    }
    const prompt = this.deps
      .ask(origin)
      .then((allowed) => {
        if (allowed) {
          this.deniedAt.delete(origin);
          this.deps.grant(origin);
        } else {
          this.deniedAt.set(origin, this.deps.clock.now());
        }
      })
      .catch((error: unknown) => console.error('[api] origin prompt failed', error))
      .finally(() => this.pending.delete(origin));
    this.pending.set(origin, prompt);
  }

  async idle(): Promise<void> {
    await Promise.all(this.pending.values());
  }
}
```

```ts
// rate-limiter.ts
import type { Clock } from '../../core/types';

const MS_PER_SECOND = 1_000;

/** 每个调用方一个令牌桶：程序出错反复提交时，挡住的是它自己，不影响别的调用方。 */
export class RateLimiter {
  private readonly buckets = new Map<string, { tokens: number; at: number }>();

  constructor(
    private readonly clock: Clock,
    private readonly limits: { perSecond: number; burst: number },
  ) {}

  take(callerId: string): boolean {
    const now = this.clock.now();
    const bucket = this.buckets.get(callerId) ?? { tokens: this.limits.burst, at: now };
    const refilled = Math.min(
      this.limits.burst,
      bucket.tokens + ((now - bucket.at) / MS_PER_SECOND) * this.limits.perSecond,
    );
    const allowed = refilled >= 1;
    this.buckets.set(callerId, { tokens: allowed ? refilled - 1 : refilled, at: now });
    return allowed;
  }
}
```

```ts
/** 在隐藏窗口里加载标签 HTML 并使用它（打印、导出 PDF 共用）；用完一定销毁。 */
export async function withLabelWindow<T>(html: string, use: (contents: WebContents) => Promise<T>): Promise<T> {
  const window = new BrowserWindow({
    show: false,
    webPreferences: { sandbox: true, contextIsolation: true, nodeIntegration: false, javascript: false },
  });
  try {
    await window.loadURL(`data:text/html;charset=utf-8,${encodeURIComponent(html)}`);
    return await use(window.webContents);
  } finally {
    if (!window.isDestroyed()) {
      window.destroy();
    }
  }
}
```
